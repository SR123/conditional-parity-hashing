theorem ConditionalParityHashing.sourceLaw_mass_on_atoms : ∀ (a : Fin 24),
  sourceLaw.mass (sourceAtoms a) = ↑(sourceWeights a) :=
fun a =>
  id
    (Eq.mpr
      (id
        (congrFun'
          (congrArg Eq
            (sum_congr (Eq.refl univ) fun x a_1 =>
              ite_congr
                ((fun x_0 x_1 => propext ((fun x_0 x_1 => Function.Injective.eq_iff atoms_injective) x_0 x_1)) x a)
                (fun a => Eq.refl ↑(sourceWeights x)) fun a => Eq.refl 0))
          ↑(sourceWeights a)))
      (of_eq_true
        (Eq.trans
          (congrFun'
            (congrArg Eq
              (Eq.trans (sum_ite_eq' univ a fun x => ↑(sourceWeights x))
                (ite_eq_left_of_eq_true (↑(sourceWeights a)) 0 (mem_univ._simp_1 a))))
            ↑(sourceWeights a))
          (eq_self ↑(sourceWeights a)))))
theorem ConditionalParityHashing.sourceLaw_zero_off_atoms : ∀ (w : Word4 4),
  (∀ (a : Fin 24), sourceAtoms a ≠ w) → sourceLaw.mass w = 0 :=
fun w hw =>
  id
    (of_eq_true
      (Eq.trans
        (congrFun'
          (congrArg Eq
            (Eq.trans
              (sum_congr (Eq.refl univ) fun x a =>
                Eq.trans
                  (ite_congr ((fun a => eq_false (hw a)) x) (fun a => Eq.refl ↑(sourceWeights x)) fun a => Eq.refl 0)
                  (ite_false (↑(sourceWeights x)) 0))
              sum_const_zero))
          0)
        (eq_self 0)))
